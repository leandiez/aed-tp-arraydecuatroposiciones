\begin{tad}{Airmiles}



	\obs{Usuarios}{\dict{id:\Tipo{\Z}}{CategoriaActual : Categoria}}

	\noindent \obs{HistorialDeTransacciones}{\seq{Transacciones}}
	\\
	\texttt{obs} Promociones : dict \{ \\
	\hspace*{1.5em} NombrePromocion: $\str$, \\
	\hspace*{1.5em} $\conj{\Tipo{FactorDePromocion},\Tipo{topeDePromocion}, \Tipo{millasRestantes}}$ \\
	\}
	\vspace{0.5cm}

	\begin{proc}{CanjearMillas}
		{\Inout AM: AirMiles, \In millasCanjeadas : \Tipo{\R}, \In idUsuario : \Tipo{\Z}}
		{}
		\requiereLargo{ AM = AM_{0} \yLuego millasDisponibles = AM_{0}.Usuarios[idUsuario]_{1} \yLuego \\ millasCanjeadas > 0 \land idUsuario \in claves(AM.Usuarios) }

		\aseguraLargo{
			\IfThenElse{ millasDisponibles \geq millasCanjeadas}
			{ \\AM.Usuarios = setKey(AM_{0}.Usuarios,idUsuario, millasDisponibles - millasCanjeadas)}
			{ \\AM.Usuarios =  AM_{0}.Usuarios}
			\\ \land \tupla{idUsuario,-1,2,millasCanjeadas,'NoPromo',1.0} \in AM.HistorialDeTransacciones
			\\
			\comentario{No estoy seguro si es mejor esto, o usar el predicado que habia planteado abajo}
		}
	\end{proc}
	\\ \\
	\comentario{Predicado para transaccion de canje. \\ La idea es tener el agregado de la transaccion 'filtrado' porque en la tupla de transaccion \\hay campos que dependen del tipo de transaccion y creo que da claridad.}
	\\
	\predLargo{TransaccionCanje}{historialTransacciones: \seq{Transacciones},idUsuario : \Tipo{\Z}, millasCanjeadas : \Tipo{\R}}
	{\tupla{idUsuario,-1,2,millasCanjeadas,'NoPromo',1.0} \in historialTransacciones}

	\vspace{0.5cm}

	\auxLargo{MillasDisponibles}{IdUsuario : \Tipo{\Z}, Historial : \seq{Transacciones}}{\Tipo{\Z}}{algo}

\end{tad}
