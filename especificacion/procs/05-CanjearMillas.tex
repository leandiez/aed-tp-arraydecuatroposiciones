\section{Proc Canjear Millas}
	\vspace{1.0cm}

	\begin{proc}{CanjearMillas}
		{\Inout AM: AirMiles, \In millasCanjeadas : \Tipo{\Z}, \In idUsuario : \Tipo{\Z}}
		{}

		\requiereLargo{ AM = AM_{0} \yLuego millasDisponibles = AM_{0}.Usuarios[idUsuario]_{1} \yLuego \\ millasCanjeadas > 0 \land idUsuario \in claves(AM.Usuarios) }

		\aseguraLargo{
       		\IfThenElse{ millasDisponibles \geq millasCanjeadas}
            { AM.Usuarios = setKey(AM_{0}.Usuarios,idUsuario, millasDisponibles - millasCanjeadas)}
            { AM.Usuarios =  AM_{0}.Usuarios}
            \yLuego todoLoDemasQuedaIgual.exe
			% que si la cantidad de millas disponibles es mayor que
			% las canjeadas, las millas disponibles van a ser las
			% millas disponibles - las canjeadas. En caso de no cumplir,
			% no se modifican las millas (quedan iguales). El resto queda igual
		}
	\end{proc}
