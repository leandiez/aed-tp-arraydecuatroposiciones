\section{FinalizarPromocion}
	\vspace{1.0cm}

	\begin{proc}{FinalizarPromocion}
		{
			\Inout AM: \Tipo{Airmiles},
			\In nombrePromo: \Tipo{\seq{char}}
		}
		{}
		\requiere{
        AM = AM0 \land nombrePromo \neq "NoPromo"
        }
		\aseguraLargo{
            AM.Usuarios = AM0.Usuarios \land
			\\
			AM.HistorialDeTransacciones = AM0.HistorialDeTransacciones \land
			\\
			AM.Promociones = DelKey(AM0.Promociones, nombrePromo)
		}
	\end{proc}
