\section{Macros para escribir en LaTeX}

Para facilitar la escritura del trabajo práctico en LaTeX, les proveemos de varias macros que les permitirán escribir especificaciones en lógica de primer orden. 1
Por ejemplo, podrán hacer uso de comandos que definen los conectores lógicos: $\land, \lor, \implica$ y también los operadores de la lógica trivaluada $\yLuego$, $\oLuego$, $\implicaLuego$.

Además, disponen de comandos para cuantificadores: $\paraTodo{x}{\mathbb{Z}}{x \geq 0 \lor x < 0}$, $\existe{x}{\mathbb{Z}}{x \geq 0}$. A estos comandos se les puede agregar un \texttt{Largo} al final para que ocupen múltiples renglones.

\begin{tad}{Airmiles}

	\obs{Usuarios}{\Diccionario{id}{PropiedadesUsuario}}
	\\
	\obs{HistorialDeTransacciones}$\seq{Transferencias}$
	\\
	\texttt{obs} Promociones : dict \{ \\
	\hspace*{1.5em} NombrePromocion: $\str$, \\
	\hspace*{1.5em} $\seq{\Tipo{FactorDePromocion}, \Tipo{estadoPromocion},\Tipo{topeDePromocion}}$ \\
	\}


	\begin{proc}{crearSistema}
		{}
		{AirMiles}
		\requiere{\Tipo{True}}
		\aseguraLargo{ \norm{res.Usuarios = 0} \land \norm{res.HistorialDeTransacciones = 0} \land res.promociones = setKey(\dict{}{},"NoPromo",\seq{1,True,0}) }
	\end{proc}

	\begin{proc}{registrarUsuario}
		{
			\Inout Usuarios: \dict{id}{PropiedadesUsuario},
			\In id: \Tipo{\Z}
		}
		{}
		\comentario{Actualiza el diccionario de usuarios con  nuevo usuario} \\
		\requiere{Usuarios0 = Usuarios \land id \notin claves(Usuarios)}
		\aseguraLargo{ Usuarios = SetKey(Usuarios0, id, \tupla{
				Millas: \tupla{0,0,0,0},
				VuelosRealizados: 0,\\
				TransferenciasRealizadas: 0,
				Categorias: Bronce,
				EsReferido: \Tipo{False}
			})}
	\end{proc}

	\begin{proc}{registrarUsuarioConReferidor}
		{
			\Inout Usuarios: \dict{id}{PropiedadesUsuario},\\
			\In idReferidor: \Tipo{Z},
			\In idUsuario: \Tipo{Z}
		}
		{}
		\requiere{Usuarios0 = Usuarios \land idReferidor \in claves(Usuarios) \land idUsuario \notin claves(U)}
		\aseguraLargo{ EsNuevoUsuarioReferido(Usuarios,Usuarios0,idUsuarios)\\ \land ReferidorEsActualizado(idReferidor,Usuarios)
		}
	\end{proc}

	\predLargo{EsNuevoUsuarioReferido}{Usuarios, Usuarios0, idUsuario}{
		\comentario{Crea un usuario Referido con 500 millas totales y disponibles} \\
		Usuarios = SetKey(Usuarios0,idUsuario, \tupla{
			Millas: \tupla{500,500,0,0},
			VuelosRealizados: 0,
			TransferenciasRealizadas: 0,
			Categorias: Bronce,
			EsReferido: True,
		})
	}


	\predLargo{ReferidorEsActualizado}{Usuarios, Usuarios0, idUsuario}{
		\comentario{Actualiza el usuario referidor en Usuarios y chequea categoria nuevamente} \\
		Usuarios = SetKey(Usuarios0,idUsuario,
		\tupla{
			\\
			Millas: \tupla{MillasTotales + 500, MillasDisponibles + 500, 0, 0},\\
			VuelosRealizados: 0,
			TransferenciasRealizadas: 0, \\
			Categorias: DarCategoria(idUsuario, Usuarios), \\
			EsReferido: Usuarios\lbrack idUsuario\rbrack.EsReferido
		})
	}

	\auxLargo{DarCategoria}
	{\In \Tipo{Z}, \In Usuarios: \Diccionario{id}{PropiedadesUsuario}}
	{Categoria}
	{
		\IfThenElse{MillasTotales < 10000}{Bronce}{ \\
			\IfThenElse{MillasTotales \leq 49999}{Plata}{ \\
				\IfThenElse{MillasTotales < 1000000}{Oro}{Platino}
			}
		}
	}


	\vspace{1.0cm}

	\auxLargo{MillasTotales}
	{\In Usuarios, \In IdUsuario}
	{\Tipo{\Z}}
	{Usuarios\lbrack IdUsuario\rbrack.millas \lbrack 0 \rbrack}

	\vspace{1.0cm}
	\auxLargo{MillasDisponibles}
	{\In Usuarios, \In IdUsuario}
	{\Tipo{\Z}}
	{Usuarios\lbrack IdUsuario\rbrack.millas \lbrack 1 \rbrack}
