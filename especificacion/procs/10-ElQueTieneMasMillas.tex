\section{ElQueTieneMasMillas}
	\vspace{1.0cm}

	\begin{proc}{ElQueTieneMasMillas}
		{
			\Inout AM: \Tipo{Airmiles},
		}
		{\Tipo{\Z}}
		\requiere{ AM = AM0 \land \lvert AM.Usuarios \rvert > 0}
		\aseguraLargo{
            \existe{i}{\Tipo{\Z}}{i \in claves(AM.Usuarios) \land MillasT otales(AM.Usuarios,i) = res} \land
            \\
            \paraTodo{j}{\Tipo{\Z}}{j \in claves(AM.Usuarios) \implica MillasT otales(AM.Usuarios,j) \leq res} \land
            \\
            AM.Usuarios = AM0.Usuarios \land 
            \\
            AM.HistorialTransacciones = AM0.HistorialTransacciones \land
            \\
            AM.Promociones = AM.Promociones0
		}
	\end{proc}
