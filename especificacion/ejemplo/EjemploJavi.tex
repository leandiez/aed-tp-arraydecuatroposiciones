\section{Macros para escribir en LaTeX}

Para facilitar la escritura del trabajo práctico en LaTeX, les proveemos de varias macros que les permitirán escribir especificaciones en lógica de primer orden. 1
Por ejemplo, podrán hacer uso de comandos que definen los conectores lógicos: $\land, \lor, \implica$ y también los operadores de la lógica trivaluada $\yLuego$, $\oLuego$, $\implicaLuego$.

Además, disponen de comandos para cuantificadores: $\paraTodo{x}{\mathbb{Z}}{x \geq 0 \lor x < 0}$, $\existe{x}{\mathbb{Z}}{x \geq 0}$. A estos comandos se les puede agregar un \texttt{Largo} al final para que ocupen múltiples renglones.

\begin{tad}{Airmiles}



	\obs{Usuarios}{\Diccionario{id}{PropiedadesUsuario}}
	\\
	\obs{HistorialDeTransacciones}$\seq{Transacciones}$
	\\
	\texttt{obs} Promociones : dict \{ \\
	\hspace*{1.5em} NombrePromocion: $\str$, \\
	\hspace*{1.5em} $\seq{\Tipo{FactorDePromocion}, \Tipo{estadoPromocion},\Tipo{topeDePromocion}}$ \\
	\}


	\begin{proc}{crearSistema}
		{}
		{AirMiles}
		\requiere{\Tipo{True}}
		\aseguraLargo{ \norm{res.Usuarios = 0} \land \norm{res.HistorialDeTransacciones = 0} \land res.promociones = setKey(\dict{}{},"NoPromo",\seq{1,True,0}) }
	\end{proc}

	\begin{proc}{registrarUsuario}
		{
			\Inout AM: Airmiles,
			\In id: \Tipo{\Z}
		}
		{}
		\comentario{Actualiza el diccionario de usuarios con  nuevo usuario} \\
		\requiere{AM = AM0 \land id \notin claves(AM.Usuarios)}
		\aseguraLargo{ AM.Usuarios = SetKey(AM0.Usuarios, id, \\
			\tupla{
				Millas: \tupla{0,0,0,0},
				VuelosRealizados: 0,\\
				TransferenciasRealizadas: 0,
				Categorias: Bronce,
				EsReferido: \Tipo{False}
			}) \land
			AM.HistorialDeTransacciones = AM0.HistorialDeTransacciones \\
			\land AM.Promociones = AM0.Promociones
		}
	\end{proc}


	\begin{proc}{registrarUsuarioConReferidor}
		{
			\Inout AM: Airmiles,\\
			\In idReferidor: \Tipo{\Z},
			\In idUsuario: \Tipo{\Z}
		}
		{}
		\requiere{AM0 = AM \land idReferidor \in claves(AM.Usuarios) \land idUsuario \notin claves(AM.Usuarios)}
		\aseguraLargo{ EsNuevoUsuarioReferido(AM,AM0: Airmiles,idUsuario: \Tipo{\Z})\\ \land ReferidorEsActualizado(AM,AM0,idReferidor) \land \\
			AM.HistorialDeTransacciones = AM0.HistorialDeTransacciones \land \\
			AM.Promociones = AM0.Promociones
		}
	\end{proc}

	\predLargo{ReferidorEsActualizado}{AM,AM0: \Tipo{AirMiles}, idReferidor: \Tipo{\Z}}{
		\comentario{Actualiza el usuario referidor en Usuarios y chequea categoria nuevamente} \\
		AM.Usuarios = SetKey(AM0.Usuarios,idUsuario,
		\tupla{
			\\
			Millas: \tupla{MillasTotales(AM.usuarios,idReferidor) + 500, \\ MillasDisponibles(AM.usuarios,idReferidor) + 500, 0, 0},\\
			VuelosRealizados: 0,
			TransferenciasRealizadas: 0, \\
			Categorias: DarCategoria(idUsuario, AM.Usuarios), \\
			EsReferido: AM0.Usuarios.EsReferido
		})
	}

	\auxLargo{DarCategoria}
	{\In id: \Tipo{\Z}, \In Usuarios: \Diccionario{id}{PropiedadesUsuario}}
	{Categoria}
	{
		\IfThenElse{MillasTotales(\Tipo{Usuarios},\Tipo{id}) < 10000}{Bronce}{ \\
			\IfThenElse{MillasTotales(\Tipo{Usuarios},\Tipo{id}) \leq 49999}{Plata}{ \\
				\IfThenElse{MillasTotales(\Tipo{Usuarios},\Tipo{id}) < 1000000}{Oro}{Platino}
			}
		}
	}
