\begin{proc}{TransferirMillas}
	{
		\Inout AM: Airmiles,\\
		\In idUsuarioPagador: \Tipo{\Z},
		\In idUsuarioBeneficiario: \Tipo{\Z},
		\In CantMillasTrans: \Tipo{\R}
	}
	{}
	\requiere{AM0 = AM \land idUsuarioPagador \in claves(AM.usuarios)\\
		\land idUsuarioBeneficiario \in claves(AM.usuarios)\\
		\land idUsuarioPagador \neq idUsuarioBeneficiario\\
		\land TieneSuficientesMillas(AM, idUsuarioPagador)
	}
	\aseguraLargo{Transaccion(AM, AM0, idUsuarioPagador, idUsuarioBeneficiario, -CantMillasTrans) \land
		Transaccion(AM, AM0, idUsuarioBeneficiario,idUsuarioPagador,  CantMillasTrans)\\
		\land AM.Promociones = AM0.Promociones \\
		\land AM.Usuarios = AM0.Usuarios
	}
	\noindent \comentarioLargo{no hice el pred Transaccion por si ya estaba hecho, ni el TieneSuficientesMillas, ¿debo poner dos transacciones en el historial?}
\end{proc}
