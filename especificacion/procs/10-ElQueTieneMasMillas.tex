\section{ElQueTieneMasMillas}
	\vspace{1.0cm}

	\begin{proc}{ElQueTieneMasMillas}
		{
			\In AM: \Tipo{Airmiles},
		}
		{\Tipo{\Z}}
		\requiere{ \norm{AM.Usuarios} > 0}
		\aseguraLargo{
            \existe{i}{\Tipo{\Z}}{i \in claves(AM.Usuarios) \yLuego MillasTotales(AM.Historial,i) = res} \land
            \\
            \paraTodo{j}{\Tipo{\Z}}{j \in claves(AM.Usuarios) \implicaLuego MillasTotales(AM.Historial,j) \leq res}
		}
	\end{proc}
